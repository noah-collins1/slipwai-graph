C() { git -c user.name=t -c user.email=t@local commit -q "$@"; }
new() { rm -rf "$R/$1"; cp -r "$R/base" "$R/$1"; cd "$R/$1"; }
setci() { python3 - "$1" <<'PY'
import json, sys
d = json.load(open("project.json")); ci = d.setdefault("ci", {})
if sys.argv[1] == "DEL": ci.pop("branch", None)
else: ci["branch"] = json.loads(sys.argv[1])
json.dump(d, open("project.json", "w"), indent=2)
PY
}
# g [VAR=val ...] : run the gate's script directly, stdout and stderr apart, exit echoed
g() { echo "\$ $* python3 delivery/scripts/check-slice-scope.py   [$(basename "$(pwd)") on $(git branch --show-current 2>/dev/null || true)$(git symbolic-ref -q HEAD >/dev/null || echo '(detached)')]"
  env "$@" python3 delivery/scripts/check-slice-scope.py >"$R/.o" 2>"$R/.e"; rc=$?
  echo "exit $rc"; echo "[stdout]"; cat "$R/.o"; echo "[stderr]"; cat "$R/.e"; echo; }
