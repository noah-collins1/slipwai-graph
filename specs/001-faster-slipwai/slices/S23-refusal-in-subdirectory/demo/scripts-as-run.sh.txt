################ /tmp/s23-demo/at-terminal.py
# a person at a terminal pressing Enter at every question: usage at-terminal.py <cwd> <slipwai> args...
import os, pty, select, subprocess, sys, time
cwd, argv = sys.argv[1], sys.argv[2:]
a, b = pty.openpty()
p = subprocess.Popen(argv, cwd=cwd, stdin=b, stdout=b, stderr=b); os.close(b)
out = bytearray(); sent = 0; end = time.monotonic() + 180
while time.monotonic() < end:
    r, _, _ = select.select([a], [], [], 1.0)
    if r:
        try: c = os.read(a, 4096)
        except OSError: break
        if not c: break
        out += c
    elif p.poll() is not None: break
    elif sent < 40: os.write(a, b"\r"); sent += 1
p.wait(); sys.stdout.write(out.decode(errors='replace')); print(f"\n[exit {p.returncode}; Enter pressed {sent} times]")

A=/home/noahc/math/slipwai-graph/slipwai; B=/tmp/s23-demo/before/slipwai
echo "##### AC-S23-10: slipwai add-service's own refusal, the factory before the slice and this one (a generated project, an uncommitted change in it)"
for S in $B $A; do
  O=/tmp/s23-demo/gen-$(basename $(dirname $S)); rm -rf $O; mkdir -p $O
  "$S" generate proj --backend python --output $O --no-init --no-install > $O.log 2>&1; echo "factory $S: generate exit $? ($(tail -1 $O.log | cut -c1-120))"

  "$S" add-service billing 2>&1 | grep -v -e '^usage:' -e '^   *\[' | sed "s#$O#<O>#g" > $O.refusal; echo "[exit ${PIPESTATUS[0]}]"; cat $O.refusal
done
