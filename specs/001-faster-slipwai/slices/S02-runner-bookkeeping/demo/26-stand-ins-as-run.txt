#!/bin/bash
# drive-park.sh <scenario> : runs the runner in the background (parking allowed), acts as the person, stops it.
s=$1; cd /tmp/s02-demo/q/shop; unset CRUISE_RUNNER CRUISE_ITERATION
export CRUISE_POLL_SECONDS=0.2 CRUISE_HARNESS_COMMAND="$(sh /tmp/s02-demo/h/mk.sh $s /tmp/s02-demo/h/$s.b)"
out=/tmp/s02-demo/h/$s/runner.out; start=$(wc -l < specs/cruise-log.jsonl)
timeout 90 python3 scripts/agents/cruise.py run </dev/null >"$out" 2>&1 & pid=$!
waitfor(){ for i in $(seq 1 150); do grep -q "$1" "$out" && [ "$(grep -c 'cruise: waiting' "$out")" -ge "$2" ] && return 0; sleep 0.2; done; echo "(timed out waiting for park $2)"; }
person(){ echo ">>> person: $*"; }
if [ "$s" = parkedit ]; then
  waitfor "parked" 1; person "while the run is parked, appends a line to scripts/check-imports.py, then: cruise.py tell go on"
  echo "# a line a person added during the park" >> scripts/check-imports.py
  python3 scripts/agents/cruise.py tell "go on"
  waitfor "parked" 2
else
  waitfor "parked" 1; person "keeps the change on purpose: cruise.py tell keep the hook file"
  python3 scripts/agents/cruise.py tell "keep the hook file"
  waitfor "parked" 2
fi
touch .specify/cruise.stop; wait $pid; echo "runner exit $?"
echo "--- runner output:"; grep -v "each iteration runs\|harness: CRUISE" "$out"
echo "--- entries this run wrote:"; tail -n +$((start+1)) specs/cruise-log.jsonl
rm -f .specify/cruise.stop

=================== dec.py
#!/usr/bin/env python3
"""dec.py <out> <spec>... : write a decisions.md. Each spec is `scope[;status]`; scope `-` means no Scope line."""
import sys
out, specs = sys.argv[1], sys.argv[2:]
text = "# Decisions\n\n"
for n, spec in enumerate(specs, 1):
    scope, _, status = spec.partition(";")
    line = "" if scope == "-" else ("- **Scope:**" + ("" if scope == "EMPTY" else " " + scope) + "\n")
    text += (f"## D{n} — question {n}\n"
             f"- **Stage:** specify · **Slice:** S02-runner-bookkeeping · **When:** 2026-10-03T10:00:00Z · **Iteration:** {n}\n"
             f"{line}"
             f"- **Question:** question {n}?\n- **Options:** a (recommended) | b\n- **Decision:** a\n- **Why:** because {n}\n"
             f"- **Decided by:** host (stage recommendation)\n- **Confidence:** high · **Would reverse if:** never\n"
             f"- **Written to:** README.md\n- **Status:** {status or 'standing'}\n\n")
open(out, "w").write(text)

=================== opens.py
"""Run scripts/agents/code_index.py health in-process and count which tracked files it opens for content."""
import runpy, subprocess, sys, os
tracked = set(subprocess.run(["git", "ls-files"], capture_output=True, text=True).stdout.split("\n"))
root = os.getcwd(); seen = []
def hook(event, args):
    if event == "open" and isinstance(args[0], str):
        rel = os.path.relpath(args[0], root) if os.path.isabs(args[0]) else args[0]
        if rel in tracked and not rel.startswith("scripts/"):
            seen.append(rel)
sys.addaudithook(hook)
sys.argv = ["scripts/agents/code_index.py", "health"]
sys.dont_write_bytecode = True
try:
    runpy.run_path("scripts/agents/code_index.py", run_name="__main__")
except SystemExit as e:
    code = e.code
else:
    code = 0
print(f"  [observer] tracked files outside scripts/ opened by this process: {len(seen)} ({len(set(seen))} distinct)" + (": " + ", ".join(sorted(set(seen))[:4]) if 0 < len(set(seen)) <= 4 else ""))
