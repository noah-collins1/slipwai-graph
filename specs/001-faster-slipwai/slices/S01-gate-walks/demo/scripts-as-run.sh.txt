import os, sys, runpy
total = 0
inside = []
def hook(event, args):
    global total
    if event == "os.scandir" and args and args[0] is not None:
        p = os.fspath(args[0])
        if isinstance(p, bytes): p = p.decode()
        a = os.path.abspath(p)
        if "/apps" in a or "/packages" in a:
            try: total += len(os.listdir(a))
            except OSError: pass
        for name in (".venv", "node_modules", "__pycache__", ".git", "target"):
            if f"/{name}/" in a + "/" and "/apps/" in a: inside.append(a)
    if event == "open" and args and str(args[0]).endswith("project.json"):
        opens.append(args[0])
opens = []
sys.addaudithook(hook)
script = sys.argv[1]; sys.argv = [script]
try: runpy.run_path(script, run_name="__main__")
except SystemExit as e: code = e.code
else: code = 0
print(f"[audit] {script}: exit {code}; entries listed under apps/ and packages/ = {total}; project.json opened {len(opens)} time(s); listings at or inside a pruned-name directory = {len(inside)}", file=sys.stderr)
for a in inside[:6]: print("[audit]   listed:", a, file=sys.stderr)

######## /tmp/s01-demo/opened.py
# run a script under an audit hook; report how many git-tracked, indexed files it opened
import os, sys, runpy, subprocess, sqlite3
tracked = set(subprocess.run(["git","ls-files","-z"],capture_output=True).stdout.decode().split("\0")) - {""}
rows = {r[0] for r in sqlite3.connect("file:.codegraph/codegraph.db?mode=ro",uri=True).execute("select path from files")}
root = os.getcwd(); seen = set()
def hook(event, args):
    if event == "open" and isinstance(args[0], (str, bytes, os.PathLike)):
        p = os.fspath(args[0]); p = p.decode() if isinstance(p, bytes) else p
        r = os.path.relpath(os.path.abspath(p), root)
        if r in tracked and r in rows and not r.startswith("scripts/"): seen.add(r)
sys.addaudithook(hook)
script = sys.argv[1]; sys.argv = [script]
try: runpy.run_path(script, run_name="__main__")
except SystemExit as e: code = e.code
print(f"[audit] exit {code}; indexed tracked files opened (outside scripts/): {len(seen)} of {len((tracked & rows) - {t for t in tracked if t.startswith('scripts/')})} {sorted(seen)[:3] if len(seen) < 4 else ''}")
